
#include <stdio.h>

int main()
{
    int n=3, r=2;
    int alloc[3][2]={{1,0},{0,1},{1,1}};
    int max[3][2]={{2,1},{1,2},{2,2}};
    int avail[2]={1,1};
    int need[3][2], finish[3]={0}, work[2];

    // Calculate Need
    for(int i=0;i<n;i++)
        for(int j=0;j<r;j++)
            need[i][j]=max[i][j]-alloc[i][j];

    // Banker's Algorithm
    for(int j=0;j<r;j++)
        work[j]=avail[j];

    printf("Safe Sequence: ");

    for(int k=0;k<n;k++)
    {
        for(int i=0;i<n;i++)
        {
            if(!finish[i] && need[i][0]<=work[0] &&
                              need[i][1]<=work[1])
            {
                printf("P%d ",i);

                for(int j=0;j<r;j++)
                    work[j]+=alloc[i][j];

                finish[i]=1;
                break;
            }
        }
    }

    // Check Deadlock
    int deadlock=0;

    for(int i=0;i<n;i++)
        if(!finish[i])
        {
            printf("\nDeadlock at P%d",i);
            deadlock=1;
        }

    if(!deadlock)
        printf("\nSystem is Safe. No Deadlock.");

    return 0;
}
