
#include <stdio.h>
#include <stdlib.h>

int main()
{
    int n, a[20], head, i, j, total = 0, temp;

    // File Allocation
    printf("FILE ALLOCATION\n");
    printf("Enter number of blocks: ");
    scanf("%d", &n);

    printf("Enter starting block and file size: ");
    int start, size;
    scanf("%d%d", &start, &size);

    printf("Allocated blocks: ");
    for(i = 0; i < size; i++)
        printf("%d ", start + i);

    // Disk Scheduling
    printf("\n\nDISK SCHEDULING\n");
    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter requests: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    // FCFS
    total = 0;
    for(i = 0; i < n; i++)
    {
        total += abs(head - a[i]);
        head = a[i];
    }
    printf("FCFS Head Movement = %d\n", total);

    // SSTF
    printf("Enter head position again: ");
    scanf("%d", &head);

    total = 0;
    for(i = 0; i < n; i++)
    {
        int min = 999, pos = -1;
        for(j = 0; j < n; j++)
        {
            if(a[j] != -1 && abs(head-a[j]) < min)
            {
                min = abs(head-a[j]);
                pos = j;
            }
        }
        total += min;
        head = a[pos];
        a[pos] = -1;
    }
    printf("SSTF Head Movement = %d\n", total);

    return 0;
}
