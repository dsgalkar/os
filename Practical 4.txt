#include <stdio.h>
#include <pthread.h>
#include <time.h>

void* task(void* arg)
{
    printf("Thread %d is running\n", *(int*)arg);
    return NULL;
}

int main()
{
    pthread_t t[4];
    int id[4];

    clock_t start = clock();

    for(int i = 0; i < 4; i++)
    {
        id[i] = i + 1;
        pthread_create(&t[i], NULL, task, &id[i]);
    }

    for(int i = 0; i < 4; i++)
        pthread_join(t[i], NULL);

    clock_t end = clock();

    printf("\nExecution Time: %.4f seconds\n",
           (double)(end-start)/CLOCKS_PER_SEC);

    return 0;
}
